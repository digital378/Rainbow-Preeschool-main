✅ UNIVERSAL FORM TRACKING SCRIPT (COPY–PASTE)

This works for:

All <form> tags

Static pages

Landing pages

Dynamically injected forms

React / vanilla / HTML forms

<script>
(function () {
  let leadEventFired = false;

  function fireLeadEvent(form) {
    if (leadEventFired) return;
    leadEventFired = true;

    window.dataLayer = window.dataLayer || [];
    window.dataLayer.push({
      event: "conversion_event_submit_lead_form",
      form_id: form.id || null,
      form_name: form.getAttribute("name") || null,
      page_path: window.location.pathname,
      page_url: window.location.href
    });

    // Reset lock after short delay (prevents duplicates)
    setTimeout(() => {
      leadEventFired = false;
    }, 3000);
  }

  document.addEventListener("submit", function (e) {
    const form = e.target;
    if (!form || form.tagName !== "FORM") return;

    // Optional: ignore search/newsletter forms if needed
    // if (form.classList.contains("ignore-lead")) return;

    fireLeadEvent(form);
  }, true);
})();
</script>
