## MOTION — use GSAP + ScrollTrigger (reuse if already on the site) + canvas-confetti.
## Wrap ALL of this in: if (!matchMedia('(prefers-reduced-motion:reduce)').matches){…}
## transform/opacity only; will-change:transform on .pc-inner.

// 1) POINTER-TRACKED 3D TILT + intra-card parallax (the core effect)
gsap.utils.toArray('.prog-card').forEach(card => {
  const inner  = card.querySelector('.pc-inner');
  const layers = card.querySelectorAll('.layer');
  const rX = gsap.quickTo(inner, 'rotationX', {duration:.4, ease:'power3.out'});
  const rY = gsap.quickTo(inner, 'rotationY', {duration:.4, ease:'power3.out'});
  card.addEventListener('pointermove', e => {
    const r = card.getBoundingClientRect();
    const px = (e.clientX - r.left)/r.width  - .5;   // -0.5..0.5
    const py = (e.clientY - r.top )/r.height - .5;
    rY(px * 16);            // up to ~8° each way
    rX(-py * 16);
    layers.forEach(l => gsap.to(l, {                 // deeper layers travel more
      x: px * (+l.dataset.depth) * .5,
      y: py * (+l.dataset.depth) * .5,
      duration:.4, ease:'power3.out'}));
  });
  card.addEventListener('pointerleave', () => {
    rX(0); rY(0);
    gsap.to(layers, {x:0, y:0, duration:.6, ease:'elastic.out(1,.4)'});
  });
});

// 2) JELLY PRESS (tactile) — pointerdown squish, release bounces back
card.addEventListener('pointerdown', () => gsap.to(inner,{scale:.96,duration:.12}));
card.addEventListener('pointerup',   () => gsap.to(inner,{scale:1,duration:.5,ease:'elastic.out(1,.4)'}));

// 3) CONFETTI burst in the card's theme colour on click/tap (canvas-confetti)
card.addEventListener('click', e => {
  const r = card.getBoundingClientRect();
  confetti({ particleCount:40, spread:55, startVelocity:28, scalar:.8,
    colors:[themeColour], origin:{ x:(r.left+r.width/2)/innerWidth,
                                   y:(r.top +r.height/3)/innerHeight }});
});

// 4) SCROLL-IN entrance (staggered, 3D flip-up)
gsap.from('.prog-card', { opacity:0, y:40, rotationX:-18, transformOrigin:'50% 100%',
  duration:.6, stagger:.09, ease:'back.out(1.6)',
  scrollTrigger:{ trigger:'.programmes-grid', start:'top 85%' }});

// 5) IDLE FLOAT — on the OUTER card (never on .pc-inner, so it won't fight the tilt)
gsap.utils.toArray('.prog-card').forEach((c,i) =>
  gsap.to(c, { y:'-=6', duration:3.5 + i*.3, yoyo:true, repeat:-1, ease:'sine.inOut' }));
