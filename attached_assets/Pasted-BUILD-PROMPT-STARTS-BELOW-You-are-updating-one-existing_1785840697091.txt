js
function stampAt(x, y, symbolId) {
  const stamp = document.createElement('img');
  stamp.src = '/stamps/' + symbolId + '.svg';
  stamp.className = 'ink-stamp';
  stamp.style.left = x + 'px';
  stamp.style.top = y + 'px';
  document.body.appendChild(stamp);
  requestAnimationFrame(() => stamp.classList.add('thump'));
  setTimeout(() => stamp.classList.add('fade'), 900);
  setTimeout(() => stamp.remove(), 1500);
  passportStrip.markCollected(symbolId); // fills the passport strip
}
