## STEP 1 — Add this adapted component file: components/ui/programme-card.tsx
# (adapted from 21st.dev "Parallax Card"; real preserve-3d + translateZ parallax)

import * as React from "react";
import { motion, useMotionValue, useSpring, useTransform } from "framer-motion";

export interface ProgrammeCardProps {
  title: string; ageLabel: string; description: string;
  imageUrl: string; href: string; themeColor: string; // hex per card
}

export function ProgrammeCard({ title, ageLabel, description, imageUrl, href, themeColor }: ProgrammeCardProps) {
  const x = useMotionValue(0), y = useMotionValue(0);
  const xs = useSpring(x, { stiffness: 300, damping: 30 });
  const ys = useSpring(y, { stiffness: 300, damping: 30 });
  const rotateX = useTransform(ys, [-0.5, 0.5], ["10deg", "-10deg"]);
  const rotateY = useTransform(xs, [-0.5, 0.5], ["-10deg", "10deg"]);
  const zImg = useTransform(ys, [-0.5, 0.5], [-14, 14]);     // photo layer travel
  const zTxt = useTransform(ys, [-0.5, 0.5], [14, -14]);     // text layer travel (opposite)

  const onMove = (e: React.MouseEvent<HTMLAnchorElement>) => {
    const r = e.currentTarget.getBoundingClientRect();
    x.set((e.clientX - r.left) / r.width - 0.5);
    y.set((e.clientY - r.top) / r.height - 0.5);
  };
  const onLeave = () => { x.set(0); y.set(0); };

  return (
    <motion.a href={href} onMouseMove={onMove} onMouseLeave={onLeave}
      style={{ rotateX, rotateY, transformStyle: "preserve-3d" }}
      className="group relative block h-[420px] w-full rounded-[24px]">
      {/* themed soft base */}
      <div className="absolute inset-0 rounded-[24px]"
challenge        style={{ background: `linear-gradient(160deg, ${themeColor}22, ${themeColor}0d)` }} />
      {/* raised inner panel */}
      <div className="relative flex h-full flex-col overflow-hidden rounded-[22px] bg-white shadow-[0_18px_40px_rgba(33,27,46,0.14)]"
        style={{ transform: "translateZ(50px)", transformStyle: "preserve-3d" }}>
        {/* PHOTO layer (deepest travel) */}
        <motion.div style={{ transform: "translateZ(30px)", translateY: zImg }}
          className="relative h-[220px] w-full overflow-hidden">
          <img src={imageUrl} alt={title} loading="lazy"
            className="h-full w-full object-cover transition-transform duration-500 group-hover:scale-[1.06]" />
          <div className="absolute inset-x-0 bottom-0 h-1/3"
            style={{ background: `linear-gradient(to top, ${themeColor}33, transparent)` }} />
          <span className="absolute right-3 top-3 rounded-full px-3 py-1 text-xs font-semibold text-white shadow-md"
            style={{ background: themeColor }}>{ageLabel}</span>
        </motion.div>
        {/* TEXT layer (opposite travel = parallax depth) */}
        <motion.div style={{ transform: "translateZ(60px)", translateY: zTxt }}
          className="flex flex-1 flex-col p-5">
          <h3 className="text-lg font-bold" style={{ color: themeColor }}>{title}</h3>
          <p className="mt-2 text-sm leading-relaxed text-neutral-600">{description}</p>
          <span className="mt-auto inline-flex items-center gap-1 pt-4 text-sm font-semibold"
            style={{ color: themeColor }}>
            Learn More <span className="transition-transform group-hover:translate-x-1">→</span>
          </span>
        </motion.div>
      </div>
    </motion.a>
  );
}
